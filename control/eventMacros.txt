automacro EnableAutolootOnLogin {
	OnCharLogIn 1
	call {
		pause 2
		do c @autoloot 100
	}
}

automacro BotCheckAlert {
	SimpleHookEvent captcha_file
	overrideNotWhenInQueue 1
	priority 0
	call {
		do eval print "\a\a\a"
		do eval system(1, "mspaint", "$.SimpleHookEventLastFile")
		log *** BOT CHECK from the server! Read the picture and type:  captcha <answer> ***
		log *** Picture saved at $.SimpleHookEventLastFile ***
	}
}

automacro EscapeOnLowHP {
	NotInMap prontera
	CurrentHP < 10%
	InInventory "Butterfly Wing" > 0
	isAlive 1
	exclusive 1
	priority 0
	timeout 5
	call {
		log HP below 10% - Butterfly Wing to Prontera for the healer.
		do is Butterfly Wing
	}
}

automacro TeleportFromHiddenAttacker {
	Eval ($::field && !$::field->isCity && scalar grep { ($_->{statuses}{EFFECTSTATE_HIDING} || $_->{statuses}{EFFECTSTATE_BURROW}) && ($_->{dmgToYou} || $_->{missedYou} || $_->{castOnToYou}) } @{$::monstersList->getItems}) ? 1 : 0
	isAlive 1
	exclusive 1
	priority 0
	timeout 5
	call {
		log A hidden monster is attacking - teleporting away.
		do tele
	}
}

automacro RebuffWhenBuffsOff {
	Eval ($::field && $::field->baseName eq $::config{lockMap} && time - $::timeout{ai}{time} > 8) ? 1 : 0
	StatusInactiveHandle EFST_BLESSING, EFST_INC_AGI
	InInventory "Butterfly Wing" > 0
	isAlive 1
	exclusive 1
	timeout 30
	call {
		log Buff wore off - Butterfly Wing to Prontera for the healer.
		do is Butterfly Wing
	}
}

automacro WingWhenOverweight {
	Eval ($::field && !$::field->isCity && !AI::inQueue('teleport') && Misc::percent_weight($::char) >= $::config{itemsMaxWeight_sellOrStore} && AI::ai_storageAutoCheck() && time - $::timeout{ai}{time} > 8) ? 1 : 0
	InInventory "Butterfly Wing" > 0
	isAlive 1
	exclusive 1
	overrideNotWhenInQueue 1
	timeout 30
	call {
		log Weight at or above &config(itemsMaxWeight_sellOrStore)% - Butterfly Wing to Prontera to store.
		do is Butterfly Wing
	}
}

automacro WingToPronteraIfElsewhere {
	Eval ($::field && $::field->baseName ne 'prontera' && $::field->baseName ne $::config{lockMap} && time - $::timeout{ai}{time} > 8) ? 1 : 0
	InInventory "Butterfly Wing" > 0
	isAlive 1
	exclusive 1
	timeout 30
	call {
		log On $.map instead of &config(lockMap) - Butterfly Wing to Prontera to use the warper.
		do is Butterfly Wing
	}
}

automacro PronteraToHuntMap {
	InMap prontera
	Eval (Misc::percent_weight($::char) < $::config{itemsMaxWeight_sellOrStore} || !AI::ai_storageAutoCheck()) ? 1 : 0
	isAlive 1
	exclusive 1
	timeout 20
	call {
		pause 2
		$berserkCount = &invamount(Berserk Potion)
		if ($berserkCount < 1) {
			log No Berserk Potion in inventory - withdrawing from storage.
			do autostorage
			pause 3
			$storageBusy = &eval(AI::inQueue('storageAuto', 'buyAuto', 'sellAuto') ? 1 : 0)
			while ($storageBusy == 1) {
				pause 1
				$storageBusy = &eval(AI::inQueue('storageAuto', 'buyAuto', 'sellAuto') ? 1 : 0)
			}
		}
		log In Prontera - visiting the healer.
		do move 162 194
		pause 1
		do talknpc 162 196 n
		pause 1
		$sitX = &rand(152, 172)
		$sitY = &rand(184, 192)
		$sitTime = &rand(60, 300)
		log Healed and buffed - walking to $sitX $sitY to sit for $sitTime seconds.
		do move $sitX $sitY
		pause &rand(6, 10)
		do sit
		pause $sitTime
		do stand
		pause 1
		log Done resting - visiting the healer again.
		do move 162 194
		pause &rand(6, 10)
		do talknpc 162 196 n
		pause 1
		log Rebuffed - taking the warper to &config(lockMap).
		do talknpc 159 196 &config(warper_steps)
	}
}

automacro QuitWhenOverweight {
	CharCurrentWeight >= 90%
	exclusive 1
	priority 0
	call {
		log Weight is at 90% or more - quitting.
		do ai off
		do quit
	}
}
