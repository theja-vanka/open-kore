automacro EnableAutolootOnLogin {
	OnCharLogIn 1
	call {
		pause 2
		do c @autoloot 100
	}
}

automacro EscapeOnLowHP {
	NotInMap prontera
	CurrentHP < 10%
	InInventory "Butterfly Wing" > 0
	isAlive 1
	exclusive 1
	priority 0
	timeout 5
	call {
		log HP below 10% - Butterfly Wing to Prontera for the healer.
		do is Butterfly Wing
	}
}

automacro TeleportFromHiddenAttacker {
	Eval $::field && !$::field->isCity && scalar grep { ($_->{statuses}{EFFECTSTATE_HIDING} || $_->{statuses}{EFFECTSTATE_BURROW}) && ($_->{dmgToYou} || $_->{missedYou} || $_->{castOnToYou}) } @{$::monstersList->getItems}
	isAlive 1
	exclusive 1
	priority 0
	timeout 5
	call {
		log A hidden monster is attacking - teleporting away.
		do tele
	}
}

automacro RebuffWhenBuffsOff {
	Eval $::field && $::field->baseName eq $::config{lockMap} && time - $::timeout{ai}{time} > 8
	StatusInactiveHandle EFST_BLESSING, EFST_INC_AGI
	InInventory "Butterfly Wing" > 0
	isAlive 1
	exclusive 1
	timeout 30
	call {
		log Buff wore off - Butterfly Wing to Prontera for the healer.
		do is Butterfly Wing
	}
}

automacro WingWhenOverweight {
	Eval $::field && !$::field->isCity && !AI::inQueue('teleport') && Misc::percent_weight($::char) >= $::config{itemsMaxWeight_sellOrStore} && AI::ai_storageAutoCheck() && time - $::timeout{ai}{time} > 8
	InInventory "Butterfly Wing" > 0
	isAlive 1
	exclusive 1
	overrideNotWhenInQueue 1
	timeout 30
	call {
		log Weight at or above &config(itemsMaxWeight_sellOrStore)% - Butterfly Wing to Prontera to store.
		do is Butterfly Wing
	}
}

automacro WingToPronteraIfElsewhere {
	Eval $::field && $::field->baseName ne 'prontera' && $::field->baseName ne $::config{lockMap} && time - $::timeout{ai}{time} > 8
	InInventory "Butterfly Wing" > 0
	isAlive 1
	exclusive 1
	timeout 30
	call {
		log On $.map instead of &config(lockMap) - Butterfly Wing to Prontera to use the warper.
		do is Butterfly Wing
	}
}

automacro PronteraToHuntMap {
	InMap prontera
	Eval Misc::percent_weight($::char) < $::config{itemsMaxWeight_sellOrStore} || !AI::ai_storageAutoCheck()
	isAlive 1
	exclusive 1
	timeout 20
	call {
		log In Prontera - visiting the healer.
		pause 2
		do move 162 194
		pause 1
		do talknpc 162 196 n
		pause 1
		log Healed and buffed - taking the warper to &config(lockMap).
		do talknpc 159 196 &config(warper_steps)
	}
}

automacro QuitWhenOverweight {
	CharCurrentWeight >= 90%
	exclusive 1
	priority 0
	call {
		log Weight is at 90% or more - quitting.
		do ai off
		do quit
	}
}
