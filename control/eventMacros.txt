automacro EnableAutolootOnLogin {
	OnCharLogIn 1
	call {
		$buffCheckReady = 0
		$huntMap = &config(lockMap)
		pause 2
		do c @autoloot 100
		# Buff list is empty right after (re)connecting until the server re-sends it; wait before checking buffs.
		pause 8
		$buffCheckReady = 1
	}
}

# Blessing or Increase AGI wore off while hunting: Butterfly Wing to Prontera; PronteraToHuntMap does the rest.
automacro RebuffWhenBuffsOff {
	InMap $huntMap
	VarValue $buffCheckReady 1
	StatusInactiveHandle EFST_BLESSING, EFST_INC_AGI
	InInventory "Butterfly Wing" > 0
	isAlive 1
	exclusive 1
	timeout 30
	call {
		log Buff wore off - Butterfly Wing to Prontera for the healer.
		do is Butterfly Wing
	}
}

# Walking to the hunting map is broken on this server (remade Izlude), so never walk there:
# on any map other than Prontera or the hunting map, Butterfly Wing to Prontera first.
automacro WingToPronteraIfElsewhere {
	NotInMap prontera
	VarValue $buffCheckReady 1
	InInventory "Butterfly Wing" > 0
	isAlive 1
	exclusive 1
	timeout 30
	call {
		if ($.map != $huntMap) {
			log On $.map instead of $huntMap - Butterfly Wing to Prontera to use the warper.
			do is Butterfly Wing
		}
	}
}

# In Prontera for any reason (login, respawn, after storage, rebuff): heal, then take the warper to the hunting map.
# Runs only after storageAuto has left the AI queue (eventMacro_notWhenInQueue), so the order is store -> heal -> warp.
automacro PronteraToHuntMap {
	InMap prontera
	VarValue $buffCheckReady 1
	isAlive 1
	exclusive 1
	timeout 20
	call {
		log In Prontera - visiting the healer.
		pause 2
		do move 162 194
		pause 1
		do talknpc 162 196 n
		pause 1
		log Healed and buffed - taking the warper to $huntMap.
		do talknpc 159 194 &config(warper_steps)
	}
}

automacro QuitWhenOverweight {
	CharCurrentWeight >= 90%
	exclusive 1
	priority 0
	call {
		log Weight is at 90% or more - quitting.
		do ai off
		do quit
	}
}
